cmake_minimum_required(VERSION 3.22.1)
project(ror_port_probe LANGUAGES C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "This probe is intended for the Android NDK toolchain.")
endif()
if(DEFINED CMAKE_ANDROID_NDK AND NOT CMAKE_ANDROID_NDK STREQUAL "")
    set(NDK_ROOT "${CMAKE_ANDROID_NDK}")
else()
    set(NDK_ROOT "${ANDROID_NDK}")
endif()
set(NATIVE_APP_GLUE "${NDK_ROOT}/sources/android/native_app_glue/android_native_app_glue.c")
if(NOT EXISTS "${NATIVE_APP_GLUE}")
    message(FATAL_ERROR "android_native_app_glue.c not found: check Android NDK installation.")
endif()
add_library(android_native_app_glue STATIC "${NATIVE_APP_GLUE}")
target_include_directories(android_native_app_glue PUBLIC "${NDK_ROOT}/sources/android/native_app_glue")
add_library(ror_port_probe SHARED native_activity.cpp)
target_compile_features(ror_port_probe PRIVATE cxx_std_17)
target_compile_options(ror_port_probe PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(ror_port_probe PRIVATE android_native_app_glue android log EGL GLESv3)
